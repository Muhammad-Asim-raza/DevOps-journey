[ REGISTRY TYPES ]
  Docker Hub:  hub.docker.com (default)
  ECR:         ACCOUNT.dkr.ecr.REGION.amazonaws.com
  GCR:         REGION-docker.pkg.dev
  ACR:         REGISTRY.azurecr.io
  GitHub:      ghcr.io/USERNAME/IMAGE
  Self-hosted: your.registry.com:5000

[ IMAGE NAMING FORMAT ]
  Official:    nginx:alpine
  Docker Hub:  USERNAME/IMAGE:TAG
  ECR:         ACCOUNT.dkr.ecr.REGION.amazonaws.com/IMAGE:TAG
  Local:       localhost:5000/IMAGE:TAG
  Custom host: registry.company.com/IMAGE:TAG

[ CORE COMMANDS ]
  docker login [REGISTRY]                = authenticate
  docker login --password-stdin           = secure login
  docker logout [REGISTRY]               = remove credentials
  docker pull IMAGE:TAG                  = download image
  docker push IMAGE:TAG                  = upload image
  docker push IMAGE --all-tags           = push all tags
  docker tag SOURCE TARGET               = add tag

[ LOGIN EXAMPLES ]
  # Docker Hub:
  docker login

  # Docker Hub with token:
  echo TOKEN | docker login --username USER --password-stdin

  # ECR:
  aws ecr get-login-password --region REGION | \
  docker login --username AWS --password-stdin REGISTRY

  # GitHub Container Registry:
  echo TOKEN | docker login ghcr.io --username USER --password-stdin

  # Self-hosted:
  docker login localhost:5000

[ TAGGING STRATEGY ]
  Semantic versioning (recommended):
  myapp:1.0.0     = exact version (use in production)
  myapp:1.0       = minor version
  myapp:latest    = most recent (AVOID in production)

  Git-based (fully reproducible):
  myapp:abc1234   = git commit SHA
  myapp:main-001  = branch + build number

  Environment-based:
  myapp:staging   = staging deployment
  myapp:prod      = production deployment

[ DOCKER HUB WORKFLOW ]
  1. docker login
  2. docker build -t USERNAME/app:v1.0 .
  3. docker push USERNAME/app:v1.0
  4. docker push USERNAME/app:latest
  On another machine:
  5. docker pull USERNAME/app:v1.0
  6. docker run USERNAME/app:v1.0

[ ECR WORKFLOW ]
  1. aws ecr create-repository --repository-name app
  2. aws ecr get-login-password | docker login ...
  3. docker tag app:v1.0 ACCOUNT.dkr.ecr.REGION.amazonaws.com/app:v1.0
  4. docker push ACCOUNT.dkr.ecr.REGION.amazonaws.com/app:v1.0
  (Configure AWS CLI in Phase 6)

[ SELF-HOSTED REGISTRY ]
  # Start registry:
  docker run -d -p 5000:5000 \
    -v registry-data:/var/lib/registry \
    registry:2

  # Push to local registry:
  docker tag myapp:v1.0 localhost:5000/myapp:v1.0
  docker push localhost:5000/myapp:v1.0

  # Registry API:
  curl localhost:5000/v2/_catalog       (list repos)
  curl localhost:5000/v2/app/tags/list  (list tags)

[ SECURITY BEST PRACTICES ]
  ✅ Use access tokens NOT passwords
  ✅ Rotate tokens regularly
  ✅ Use specific version tags in production
  ✅ Enable image scanning (ECR has this)
  ✅ Use lifecycle policies to clean old images
  ✅ Private repos for internal images
  ✅ Use --password-stdin not -p flag
  ❌ Never use :latest in production
  ❌ Never store credentials in code
