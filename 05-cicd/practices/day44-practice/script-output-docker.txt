[ IMAGE TAGGING STRATEGY ]
  git SHA (immutable):
  sha-abc1234 = exact commit
  Never changes, always traceable

  semantic version (release):
  v1.2.3 = human-readable release

  latest (convenience):
  latest = most recent main build
  NEVER use in production deployments!

  PR builds:
  pr-42 = PR number
  Test without polluting production tags

[ BUILD ARGS PATTERN ]
  In workflow:
  build-args: |
    APP_VERSION=${{ steps.ver.outputs.version }}
    GIT_COMMIT=${{ github.sha }}
    BUILD_TIME=$(date -u)
    GIT_BRANCH=${{ github.ref_name }}

  In Dockerfile:
  ARG APP_VERSION=dev
  ENV APP_VERSION=${APP_VERSION}
