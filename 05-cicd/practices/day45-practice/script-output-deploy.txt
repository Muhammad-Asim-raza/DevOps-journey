[ DEPLOYMENT STRATEGIES ]

  Recreate:    stop old → deploy new → start
               Downtime: YES
               Use: dev environments

  Rolling:     update 1 instance at a time
               Downtime: NO
               Rollback: slow
               Use: stateless APIs

  Blue-Green:  deploy to idle env, switch
               Downtime: NO
               Rollback: instant (switch back)
               Use: critical services

  Canary:      5% → monitor → 100%
               Downtime: NO
               Rollback: instant (0% traffic)
               Use: high-risk changes

[ SSH DEPLOYMENT PATTERN ]
  uses: appleboy/ssh-action@master
  with:
    host: ${{ secrets.PROD_HOST }}
    username: ${{ secrets.PROD_USER }}
    key: ${{ secrets.PROD_SSH_KEY }}
    script: |
      docker pull $IMAGE
      docker-compose up -d --no-deps app

[ HEALTH CHECK PATTERN ]
  MAX_WAIT=60
  ELAPSED=0
  while [ $ELAPSED -lt $MAX_WAIT ]; do
    HTTP=$(curl -s -o /dev/null -w '%{http_code}' /health)
    [ $HTTP = '200' ] && break
    sleep 5; ELAPSED=$((ELAPSED+5))
  done

[ ROLLBACK PATTERN ]
  Save current image before deploying:
  ROLLBACK_TAG=$(docker inspect app --format '{{.Config.Image}}')

  On failure:
  docker rm -f app
  docker run -d --name app $ROLLBACK_TAG
