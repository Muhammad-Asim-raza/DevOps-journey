[ BRANCH COMMANDS ]
  git branch                  = list branches
  git branch name             = create branch
  git branch -d name          = delete (safe)
  git branch -D name          = delete (force)
  git branch -a               = list all + remote
  git branch -m old new       = rename

[ SWITCHING BRANCHES ]
  git switch branch-name      = switch (modern)
  git switch -c new-branch    = create and switch
  git checkout branch-name    = switch (old way)
  git checkout -b new-branch  = create and switch

[ MERGING ]
  git merge branch-name       = merge into current
  git merge --no-ff branch    = always create merge commit
  git merge --abort           = cancel merge
  git merge --squash branch   = squash all commits

[ CONFLICT RESOLUTION ]
  1. git merge branch-name    (triggers conflict)
  2. Edit conflicted files
  3. Remove conflict markers: <<<<< ===== >>>>>
  4. git add resolved-file
  5. git commit

[ BRANCH NAMING CONVENTIONS ]
  feature/feature-name        = new features
  fix/bug-description         = bug fixes
  hotfix/critical-fix         = emergency fixes
  release/version-number      = release prep
  chore/task-description      = maintenance
  docs/doc-name               = documentation
